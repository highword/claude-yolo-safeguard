    **Step 1:** Create `src/hook/entry.ts` — The main entry point that Claude Code spawns as a child process. Structure:

    ```typescript
    import * as fs from "node:fs";
    import { loadConfig } from "../config/loader";
    import { analyzeCommand } from "../pipeline/index";
    import { applyAllowList } from "../decision/allow-list";
    import { makeDecision } from "../decision/decide";
    import { formatHookOutput } from "../decision/format";
    import { writeAuditLog } from "../decision/logger";
    import type { ClaudeCodeHookEvent } from "../types/hook";

    try {
      // Step 1: Read stdin synchronously (per RESEARCH.md Pattern 1)
      const raw = fs.readFileSync(0, "utf8");
      const event: ClaudeCodeHookEvent = JSON.parse(raw);

      // Step 2: Route by tool type (per D-65: only Bash gets full analysis)
      if (event.tool_name !== "Bash") {
        process.exit(0);
      }

      // Step 3: Extract command, bail if missing
      const command = event.tool_input?.command;
      if (!command) {
        process.exit(0);
      }

      // Step 4: Load config (3-layer merge)
      const cwd = event.cwd || process.cwd();
      const config = loadConfig(cwd);

      // Step 5: Analyze command through pipeline
      const analysis = analyzeCommand(command, config.customRules.length > 0 ? config.customRules : undefined);

      // Step 6: Apply allow-list
      const filtered = applyAllowList(analysis.matches, config.allowList, command);

      // Step 7: Make decision
      const decision = makeDecision(filtered, config.severityActions);

      // Step 8: Audit log (fire-and-forget, fail-open internally)
      writeAuditLog(decision, config.logging, {
        command,
        cwd,
        sessionId: event.session_id,
      });

      // Step 9: Format and output
      const { output, exitCode } = formatHookOutput(decision);
      if (output) {
        process.stdout.write(output);
      }
      process.exit(exitCode);
    } catch {
      // D-66: Any uncaught error = fail-open (allow operation)
      process.exit(0);
    }
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| stdin (Claude Code -> hook) | Untrusted JSON input from parent process pipe |
| stdout (hook -> Claude Code) | Decision output consumed by Claude Code |
| filesystem (config files) | User config files read during hook execution |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-01 | Tampering | src/hook/entry.ts stdin | mitigate | JSON.parse inside try-catch; malformed input triggers fail-open exit 0, never crashes |
| T-04-02 | Denial of Service | src/hook/entry.ts | mitigate | Global try-catch ensures any exception results in exit 0; hook never hangs or blocks user workflow |
| T-04-03 | Tampering | src/decision/format.ts output | accept | Output is constructed from internal Decision object, not user-controlled strings directly; systemMessage uses decision.message which is built from rule descriptions (trusted data) |
| T-04-04 | Information Disclosure | src/hook/entry.ts | accept | Hook only outputs rule IDs and descriptions (public info); no secrets exposed in stdout JSON |
</threat_model>
