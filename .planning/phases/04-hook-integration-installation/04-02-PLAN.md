    **Step 2:** Create `src/cli/init.ts` — The CLI entry point invoked via `npx claude-yolo-safeguard init`:

    ```typescript
    import * as path from "node:path";
    import { getSettingsPath, buildHookCommand, registerHook } from "./settings";
    import { getHookTargetPath, resolveHookSource, deployHook } from "./deploy";

    function main(): void {
      // Parse args: check for --project flag
      const args = process.argv.slice(2);
      const isProject = args.includes("--project");
      const mode = isProject ? "project" : "global";

      console.log(`claude-yolo-safeguard: Installing in ${mode} mode...`);

      // Step 1: Resolve source hook.cjs
      const sourcePath = resolveHookSource();
      if (!require("node:fs").existsSync(sourcePath)) {
        console.error(`Error: hook.cjs not found at ${sourcePath}`);
        process.exit(1);
      }

      // Step 2: Deploy hook to target directory
      const targetPath = getHookTargetPath(mode);
      deployHook(sourcePath, targetPath);
      console.log(`  Hook deployed to: ${targetPath}`);

      // Step 3: Register in settings.json
      const settingsPath = getSettingsPath();
      const hookCommand = buildHookCommand(targetPath);
      registerHook(settingsPath, hookCommand);
      console.log(`  Hook registered in: ${settingsPath}`);

      // Step 4: Success message
      console.log("");
      console.log("Done! yolo-safeguard is now active.");
      console.log("Claude Code will intercept dangerous Bash commands automatically.");
      console.log("");
      console.log("To verify: ask Claude to run 'rm -rf /' — it should be blocked.");
    }

    main();
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Filesystem write (installer -> ~/.claude/) | Installer writes to user's Claude Code config directory |
| Filesystem write (installer -> settings.json) | Installer modifies Claude Code's settings file |
| User input (CLI args) | --project flag from command line |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-05 | Tampering | src/cli/settings.ts | mitigate | Backup settings.json to .bak before modification; validate JSON before write with JSON.parse round-trip |
| T-04-06 | Tampering | src/cli/deploy.ts | mitigate | Validate target path starts with os.homedir() or process.cwd() + .claude/; reject traversal attempts |
| T-04-07 | Denial of Service | src/cli/settings.ts | mitigate | Read existing settings; never truncate file without valid replacement; backup ensures recovery |
| T-04-08 | Elevation of Privilege | src/cli/deploy.ts | accept | Hook runs as same user; no privilege escalation possible since Node child process inherits parent permissions |
| T-04-09 | Information Disclosure | src/cli/init.ts | accept | Only prints file paths to stdout (user's own paths); no secrets involved |
