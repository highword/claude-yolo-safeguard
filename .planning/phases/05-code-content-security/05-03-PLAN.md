    ```typescript
    // Route by tool type
    if (event.tool_name === "Write" || event.tool_name === "Edit") {
      // D-74: Write analyzes full content; Edit analyzes only new_string
      const content = event.tool_name === "Write"
        ? event.tool_input?.content
        : event.tool_input?.new_string;

      if (!content) {
        return { output: "", exitCode: 0 };
      }

      const cwd = event.cwd || process.cwd();
      const config = loadConfig(cwd);
      const filePath = event.tool_input?.file_path;

      // Content scanning (includes size guard, quick-reject, entropy, path adjustment)
      const scanResult = scanContent(content, filePath);
      if (scanResult.skipped || scanResult.matches.length === 0) {
        return { output: "", exitCode: 0 };
      }

      // Allow-list filtering (pass filePath as command for path-based allow-list matching)
      const filtered = applyAllowList(scanResult.matches, config.allowList, filePath);

      // Make graduated decision (reuse Phase 3 engine)
      const decision = makeDecision(filtered, config.severityActions);

      // Audit log
      writeAuditLog(decision, config.logging, {
        command: `${event.tool_name}: ${filePath || "unknown"}`,
        cwd,
        sessionId: event.session_id,
      });

      // Format and return
      return formatHookOutput(decision);
    }

    if (event.tool_name !== "Bash") {
      // Unknown tool: fail-open
      return { output: "", exitCode: 0 };
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| stdin JSON -> processHookEvent() | Untrusted hook event with tool_input containing arbitrary content |
| tool_input.content -> scanContent() | File content of any size passed to scanner |
| tool_input.file_path -> severity adjustment | Path used for security-relevant severity decision |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-06 | Spoofing | tool_input.file_path manipulation | mitigate | Path-based downgrade only affects HIGH->MEDIUM (warn); CRITICAL (secrets) never downgraded regardless of path; worst case: attacker gets a warning instead of block for non-secret patterns |
| T-05-07 | Denial of Service | Large content in tool_input.content | mitigate | 500KB size guard returns immediately with fail-open; content.length check is O(1) |
| T-05-08 | Information Disclosure | Audit log writes content details | accept | Audit log is local-only (user's machine); no network exfiltration; content details help user understand what was blocked |
| T-05-09 | Tampering | Malformed JSON in stdin | mitigate | try/catch around JSON.parse returns fail-open `{ output: "", exitCode: 0 }` — never crashes |
</threat_model>
