    ```typescript
    import type { Decision } from "../types/decision";
    import type { HookOutput } from "../types/hook";

    export interface FormattedOutput {
      output: string;   // JSON string for stdout (empty string if no output needed)
      exitCode: number; // 0 = allow, 2 = block
    }

    /**
     * Format a Decision into Claude Code hook protocol output.
     * Per D-62: exit 0 = allow. Per D-63: exit 2 = block with required reason JSON.
     * Per D-64: warn = exit 0 + advisory JSON.
     *
     * @param decision - Decision from makeDecision()
     * @returns { output: string, exitCode: number }
     */
    export function formatHookOutput(decision: Decision): FormattedOutput {
      // Allow path or silent actions
      if (decision.action === "off" || decision.action === "log" || decision.matchedRules.length === 0) {
        return { output: "", exitCode: 0 };
      }

      const primary = decision.matchedRules[0];

      if (decision.action === "block") {
        const hookOutput: HookOutput = {
          decision: "block",
          reason: decision.message,
          rule: primary?.rule.id,
          severity: decision.severity,
          category: primary?.rule.category,
          suggestion: decision.suggestion || undefined,
          matchedPatterns: decision.matchedRules.map(m => m.rule.id),
        };
        // Remove undefined fields before serialization
        return { output: JSON.stringify(stripUndefined(hookOutput)), exitCode: 2 };
      }

      if (decision.action === "warn") {
        const hookOutput: HookOutput = {
          decision: "allow",
          reason: decision.message,
          rule: primary?.rule.id,
          severity: decision.severity,
          category: primary?.rule.category,
          suggestion: decision.suggestion || undefined,
          matchedPatterns: decision.matchedRules.map(m => m.rule.id),
        };
        return { output: JSON.stringify(stripUndefined(hookOutput)), exitCode: 0 };
      }

      // Fallback (should not reach here)
      return { output: "", exitCode: 0 };
    }
    ```

    Helper: `stripUndefined(obj)` — returns a new object with keys whose values are undefined removed (so JSON.stringify won't include them as null).

    **Step 2: Implement `src/decision/logger.ts`:**

    ```typescript
    import * as fs from "node:fs";
    import * as os from "node:os";
    import * as path from "node:path";
    import type { Decision } from "../types/decision";
    import type { LoggingConfig } from "../types/config";

    export interface AuditContext {
      command: string;
      cwd?: string;
      sessionId?: string;
    }

    /**
     * Write a decision to the JSONL audit log.
     * Per D-55: Synchronous appendFileSync, fail-open on errors.
     * Per D-54: Block/warn = detailed record; allow = minimal record.
     * Per D-56: Single-backup rotation when file exceeds maxSizeMb.
     *
     * @param decision - Decision from makeDecision()
     * @param config - LoggingConfig from merged config
     * @param context - Command string and optional cwd/sessionId
     */
    export function writeAuditLog(
      decision: Decision,
      config: LoggingConfig,
      context: AuditContext,
    ): void {
      if (!config.enabled) return;

      try {
        const logPath = expandPath(config.path);
        ensureDirectory(logPath);
        rotateIfNeeded(logPath, config.maxSizeMb);

        const record = buildRecord(decision, context);
        fs.appendFileSync(logPath, JSON.stringify(record) + "\n");
      } catch {
        // Fail-open: never block a command because logging broke (D-55)
      }
    }
    ```

    Helper functions:
    - `expandPath(p: string): string` — replaces leading `~` with `os.homedir()`. Returns path unchanged if no `~` prefix.
    - `ensureDirectory(filePath: string): void` — `fs.mkdirSync(path.dirname(filePath), { recursive: true })`. Wrapped in try/catch (fail-open).
    - `rotateIfNeeded(logPath: string, maxSizeMb: number): void` — if file exists and `fs.statSync(logPath).size > maxSizeMb * 1024 * 1024`, rename to `logPath + ".1"` (overwrite existing .1).
    - `buildRecord(decision, context)`:
      - If decision.action is "block" or "warn": return `{ timestamp: decision.timestamp, action: decision.action, severity: decision.severity, command: context.command, cwd: context.cwd, sessionId: context.sessionId, rules: decision.matchedRules.map(m => ({ id: m.rule.id, matched: m.matchedText, description: m.rule.description })), suggestion: decision.suggestion }`
      - Otherwise (log/off/allow): return `{ timestamp: decision.timestamp, action: decision.action, command: context.command }`

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| decision -> stdout | Hook output JSON crosses back to Claude Code process |
| decision -> filesystem | Audit log writes to user's filesystem |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-04 | I (Information Disclosure) | logger.ts audit log | mitigate | Log file written to user-private directory (~/.config/yolo-safeguard/); no world-readable permissions. Commands logged may contain sensitive args — this is acceptable for security audit purposes. |
| T-03-05 | D (Denial of Service) | logger.ts rotation | mitigate | Single rotation backup (.1) limits disk usage to 2x maxSizeMb. Default 10MB = max 20MB total. |
| T-03-06 | T (Tampering) | format.ts exitCode | accept | Exit code is produced internally from severity mapping; no external input can force a specific exit code. Severity-to-action map is only escalatable by project config (loader.ts enforces). |
| T-03-07 | D (Denial of Service) | logger.ts write failure | mitigate | All fs operations wrapped in try/catch; fail-open ensures logging errors never block command execution. |
</threat_model>
